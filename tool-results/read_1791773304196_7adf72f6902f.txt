   177→function specSummaryText(spec: CreativeJobSpec): string[] {
   178→  const bits: string[] = [];
   179→  bits.push(`মান: ${spec.quality === 'draft' ? 'ড্রাফট (দ্রুত)' : spec.quality === 'high' ? 'উচ্চ' : 'স্ট্যান্ডার্ড'}`);
   180→  if (spec.language) bits.push(`ভাষা: ${LANGUAGE_LABELS[spec.language] ?? spec.language}`);
   181→  if (spec.videoEngine) bits.push(`ইঞ্জিন: ${spec.videoEngine === 'ai_diffusion' ? 'AI (CogVideoX-3)' : 'ক্যামেরা-মোশন'}`);
   182→  if (spec.filmMode) bits.push(`ফিল্ম-মোড: ${FILM_MODE_LABELS[spec.filmMode] ?? spec.filmMode}`);
   183→  if (spec.mood && MOOD_LABELS[spec.mood as MusicMood]) bits.push(`মুড: ${MOOD_LABELS[spec.mood as MusicMood]}`);
   184→  else if (spec.mood) bits.push(`মুড: ${String(spec.mood)}`);
   185→  if (typeof spec.tempoBpm === 'number') bits.push(`টেম্পো: ${toBnDigits(spec.tempoBpm)} BPM`);
   186→  if (spec.aspect) bits.push(`অনুপাত: ${toBnDigits(spec.aspect)}`);
   187→  if (spec.durationSec) bits.push(`দৈর্ঘ্য: ${toBnDigits(spec.durationSec)} সেকেন্ড`);
   188→  if (spec.fps) bits.push(`FPS: ${toBnDigits(spec.fps)}`);
   189→  if (spec.sound) bits.push(`শব্দ: ${{ none: 'নীরব', ambient: 'পরিবেশ-শব্দ', narration: 'ভয়েসওভার', uploaded_song: 'আপলোড করা গান' }[spec.sound]}`);
   190→  if (spec.voice) bits.push(`কণ্ঠ: ${spec.voice}`);
   191→  if (spec.subtitles) bits.push('সাবটাইটেল: হ্যাঁ');
   192→  if (spec.format) bits.push(`ফরম্যাট: ${IMAGE_FORMAT_LABELS[spec.format]}`);
   193→  return bits;
   194→}

   436→function StepList({ view }: { view: ProductionView }) {

   442→            {step.status === 'done' ? (
   443→              <CheckCircle2 className="size-4 text-emerald-600 dark:text-emerald-400" />

   446→                <Loader2 className="size-4 animate-spin text-emerald-600 dark:text-emerald-400" />
   447→              </span>
   448→            ) : step.status === 'failed' ? (
   449→              <XCircle className="size-4 text-red-500" />
   450→            ) : step.status === 'skipped' ? (
   451→              <Ban className="size-4 text-muted-foreground" />
   452→            ) : (
   453→              <CircleDashed className="size-4 text-muted-foreground/60" />
   454→            )}
   455→          </span>
   456→          <span className="min-w-0 flex-1">
   457→            <span className={cn('block leading-tight', step.status === 'pending' && 'text-muted-foreground/70')}>
   458→              {step.title}
   459→            </span>
   460→            {step.detail && <span className="block text-xs text-muted-foreground">{step.detail}</span>}
   461→            {step.error && <span className="block text-xs text-red-500">{step.error}</span>}
   462→          </span>
   463→          {step.status === 'done' && step.durationMs != null && step.durationMs > 0 && (
   464→            <span className="mt-0.5 shrink-0 text-[11px] tabular-nums text-muted-foreground/80">
   465→              {toBnDigits((step.durationMs / 1000).toFixed(1))} সেকেন্ড
   466→            </span>
   467→          )}
   468→        </li>
   469→      ))}
   470→    </ol>
   471→  );
   472→}

   478→export function ProductionCard({ productionId }: { productionId: string }) {
   479→  const view = useProductions(s => s.productions[productionId]);
   480→  const ensureFetched = useProductions(s => s.ensureFetched);
   481→  const subscribe = useProductions(s => s.subscribe);
   482→  const confirmProduction = useProductions(s => s.confirmProduction);
   483→  const cancelProduction = useProductions(s => s.cancelProduction);
   484→  const reviseProduction = useProductions(s => s.reviseProduction);
   485→
   486→  const [editedSpec, setEditedSpec] = useState<CreativeJobSpec | null>(null);
   487→  const [isConfirming, setIsConfirming] = useState(false);
   488→  const [isRevising, setIsRevising] = useState(false);
   489→  const [revisionId, setRevisionId] = useState<string | null>(null);
   490→
   491→  useEffect(() => {
   492→    void ensureFetched(productionId);
   493→  }, [productionId, ensureFetched]);
   494→
   495→  useEffect(() => {
   496→    if (view) subscribe(productionId);
   497→  }, [view, productionId, subscribe]);
   498→
   499→  if (!view) {
   500→    return (
   501→      <div className="rounded-xl border bg-card p-3 text-sm text-muted-foreground shadow-sm">
   502→        <Loader2 className="mr-2 inline size-4 animate-spin" aria-hidden="true" />
   503→        প্রোডাকশন লোড হচ্ছে…
   504→      </div>
   505→    );
   506→  }
   507→
   508→  const typeMeta = TYPE_META[view.type] ?? TYPE_META.image;
   509→  const TypeIcon = typeMeta.icon;
   510→  const statusMeta = STATUS_META[view.status];
   511→  const isAwaiting = view.status === 'awaiting_confirmation';
   512→  const isActive = ['queued', 'running', 'validating', 'cancelling'].includes(view.status);
   513→  const isDone = ['completed', 'partial'].includes(view.status);

   517→  if (view.resultJson) {
   518→    try {
   519→      resultSummary = JSON.parse(view.resultJson);
   520→    } catch {
   521→      resultSummary = null;
   522→    }
   523→  }
   524→
   525→  const spec = editedSpec ?? view.spec;
   526→

   529→    const ok = await confirmProduction(productionId, editedSpec ? (editedSpec as unknown as Record<string, unknown>) : undefined);
   530→    setIsConfirming(false);
   531→    if (ok) toast({ title: 'প্রোডাকশন শুরু হয়েছে', description: view.title });
   532→  };
   533→
   534→  const handleRevise = async () => {
   535→    setIsRevising(true);
   536→    const revision = await reviseProduction(productionId, (editedSpec ?? view.spec) as unknown as Record<string, unknown>);
   537→    setIsRevising(false);
   538→    if (revision) {
   539→      setRevisionId(revision.id);
   540→      toast({ title: 'সংশোধন-প্রস্তাব তৈরি হয়েছে', description: 'নিচের নতুন কার্ডে নিশ্চিত করুন' });
   541→    }
   542→  };
   543→
   544→  return (
   545→    <div className="overflow-hidden rounded-xl border bg-card shadow-sm">
   546→      {/* হেডার */}
   547→      <div className="flex items-start gap-3 border-b bg-muted/40 px-4 py-3">
   548→        <span className="mt-0.5 flex size-9 shrink-0 items-center justify-center rounded-lg bg-emerald-600/10 text-emerald-700 dark:text-emerald-400">
   549→          <TypeIcon className="size-5" aria-hidden="true" />
   550→        </span>
   551→        <div className="min-w-0 flex-1">
   552→          <div className="flex flex-wrap items-center gap-2">
   553→            <p className="min-w-0 truncate text-sm font-semibold">{view.title}</p>
   554→            <Badge variant="outline" className={cn('shrink-0 text-[11px]', statusMeta.className)}>
   555→              {statusMeta.label}
   556→            </Badge>
   557→          </div>
   558→          <p className="mt-0.5 text-xs text-muted-foreground">{typeMeta.label}</p>
   559→        </div>
   560→      </div>
   561→
   562→      <div className="space-y-4 px-4 py-3.5">
   563→        {/* নিশ্চিতকরণ অবস্থা */}
   564→        {isAwaiting && (
   565→          <div className="space-y-3.5">
   566→            <div className="flex flex-wrap gap-1.5">
   567→              {specSummaryText(spec).map(bit => (
   568→                <Badge key={bit} variant="secondary" className="text-[11px] font-normal">
   569→                  {bit}
   570→                </Badge>

   572→            </div>

   584→            <p className="flex items-start gap-1.5 text-xs text-muted-foreground">
   585→              <PencilRuler className="mt-0.5 size-3.5 shrink-0" aria-hidden="true" />
