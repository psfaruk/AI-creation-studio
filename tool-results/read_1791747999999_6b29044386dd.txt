   117→   117→function stringifyCapability(value: unknown): string {
   118→   118→  let s: string;
   119→   119→  if (Array.isArray(value)) s = value.map(String).join(', ');
   120→   120→  else if (typeof value === 'string') s = value;
   121→   121→  else s = String(value);
   122→   122→  return s.length > 24 ? `${s.slice(0, 23)}…` : s;
   123→   123→}

   143→   143→function buildHighlightedCaps(caps: Record<string, unknown>): HighlightedCap[] {
   144→   144→  const out: HighlightedCap[] = [];
   145→   145→
   146→   146→  if (typeof caps.task_type === 'string' && caps.task_type) {
   147→   147→    out.push({
   148→   148→      key: 'task_type',
   149→   149→      render: () => ({ label: 'কাজের ধরন', value: caps.task_type as string }),
   150→   150→    });
   151→   151→  }
   152→   152→
   153→   153→  const langs = caps.supported_languages;
   154→   154→  if (langs && typeof langs === 'object' && !Array.isArray(langs)) {
   155→   155→    for (const [code, val] of Object.entries(langs as Record<string, unknown>)) {
   156→   156→      out.push({
   157→   157→        key: `lang-${code}`,
   158→   158→        render: () => ({
   159→   159→          label: LANG_CODE_LABEL[code] ?? code,
   160→   160→          value: String(val),
   161→   161→          tone: String(val).includes('দুর্বল') || String(val).includes('আংশিক') ? 'warn' : 'ok',
   162→   162→        }),
   163→   163→      });
   164→   164→    }
   165→   165→  } else if (typeof langs === 'string' && langs) {
   166→   166→    out.push({
   167→   167→      key: 'supported_languages',
   168→   168→      render: () => ({ label: 'ভাষা', value: langs }),
   169→   169→    });
   170→   170→  }
   171→   171→
   172→   172→  if (typeof caps.requires_gpu === 'boolean') {
   173→   173→    out.push({
   174→   174→      key: 'requires_gpu',
   175→   175→      render: () => ({
   176→   176→        label: 'GPU',
   177→   177→        value: caps.requires_gpu ? 'প্রয়োজন' : 'লাগে না',
   178→   178→        tone: caps.requires_gpu ? 'warn' : 'ok',
   179→   179→      }),
   180→   180→    });
   181→   181→  }
   182→   182→
   183→   183→  if (typeof caps.max_duration_sec === 'number') {
   184→   184→    out.push({
   185→   185→      key: 'max_duration_sec',
   186→   186→      render: () => ({ label: 'সর্বোচ্চ দৈর্ঘ্য', value: `${toBnDigits(caps.max_duration_sec as number)} সেকেন্ড` }),
   187→   187→    });
   188→   188→  }
   189→   189→
   190→   190→  if (typeof caps.generation_time === 'string' && caps.generation_time) {
   191→   191→    out.push({
   192→   192→      key: 'generation_time',
   193→   193→      render: () => ({ label: 'উৎপাদন-সময়', value: caps.generation_time as string }),
   194→   194→    });
   195→   195→  }
   196→   196→
   197→   197→  if (typeof caps.fallback_tool === 'string' && caps.fallback_tool) {
   198→   198→    out.push({
   199→   199→      key: 'fallback_tool',
   200→   200→      render: () => ({ label: 'বিকল্প-টুল', value: caps.fallback_tool as string }),
   201→   201→    });
   202→   202→  }
   203→   203→
   204→   204→  return out;
   205→   205→}
