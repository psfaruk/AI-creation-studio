   560→        <p className="flex items-start gap-1.5 rounded-lg border border-emerald-600/30 bg-emerald-600/5 px-3 py-2 text-xs leading-relaxed text-emerald-800 dark:text-emerald-300">
   561→          <ArrowRightLeft className="mt-0.5 size-3.5 shrink-0" aria-hidden="true" />
   562→          <span>
   563→            <span className="font-semibold">বিকল্প: </span>
   564→            {row.fallback}
   565→          </span>
   566→        </p>
   567→      )}
   568→
   569→      {/* বাধ্যতামূলক মেটা (§১০) */}
   570→      <ul className="space-y-1.5" aria-label={`${mode.title} — মোড-মেটা`}>
   571→        <ModeMetaRow icon={Languages} label="ভাষা" value={mode.languages} />
   572→        <ModeMetaRow icon={Timer} label="সর্বোচ্চ দৈর্ঘ্য" value={mode.maxDuration} />
   573→        <ModeMetaRow icon={Cog} label="ইঞ্জিন" value={ENGINE_LABELS[mode.engine]} />
   574→        <ModeMetaRow icon={Clock3} label="আনুমানিক সময়" value={mode.eta} />
   575→        <ModeMetaRow icon={CircuitBoard} label="GPU" value={mode.gpu} />
   576→      </ul>
   577→
   578→      {/* অ্যাকশন */}
   579→      <div className="mt-auto">
   580→        {isOff ? (
   581→          <p className="flex items-start gap-1.5 text-xs leading-relaxed text-muted-foreground">
   582→            <ShieldCheck className="mt-0.5 size-3.5 shrink-0" aria-hidden="true" />
   583→            এই মোড এই পরিবেশে বন্ধ (লাইসেন্সে নয়) — কারণ ও বিকল্প উপরে; রেজিস্ট্রির ম্যাট্রিক্সে বিস্তারিত।
   584→          </p>
   585→        ) : (
   586→          <Button
   587→            type="button"
   588→            onClick={() => onStart(mode.draft ?? '')}
   589→            className="h-11 w-full rounded-xl bg-emerald-600 text-white shadow-sm hover:bg-emerald-700 focus-visible:ring-emerald-600/40 dark:bg-emerald-600 dark:hover:bg-emerald-700"
   590→          >
   591→            <PlayCircle className="size-4" aria-hidden="true" />
   592→            শুরু করুন
   593→          </Button>
   594→        )}
   595→      </div>
   596→    </Card>
   597→  );
   598→}
