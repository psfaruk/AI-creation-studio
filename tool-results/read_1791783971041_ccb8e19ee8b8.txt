  1278→          if (!avSyncOk) problems.push(`A/V স্ট্রিম-দৈর্ঘ্যে অসমলয় (ভিডিও ${vd.toFixed(2)} / অডিও ${ad.toFixed(2)} সে.)`);
  1279→        }
  1280→      } catch {
  1281→        // অ-বাধামূলক
  1282→      }
  1283→    }
  1284→
  1285→    const honestLabel =
  1286→      engine === 'ai_diffusion'
  1287→        ? aiShots === shots.length
  1288→          ? `সব ${shots.length}টি শট প্রকৃত AI (${AI_VIDEO_MODEL}) — ক্লাউড ডিফিউশন ভিডিও`
  1289→          : `${aiShots}/${shots.length} শট প্রকৃত AI (${AI_VIDEO_MODEL}); বাকি ${shots.length - aiShots}টি স্টোরিবোর্ড+ক্যামেরা-মোশন ফলব্যাক`
  1290→        : 'cinematic_motion ইঞ্জিন — AI স্টোরিবোর্ড + ক্যামেরা-মোশন (প্রকৃত ডিফিউশন ভিডিও নয়)';
  1291→
  1292→    // SRT artifact
  1293→    if (srtContent) {
  1294→      await ctx.registerArtifact({
  1295→        buffer: Buffer.from(srtContent, 'utf-8'),
  1296→        kind: 'text',
  1297→        mimeType: 'application/x-subrip',
  1298→        fileName: `${sanitize(title).slice(0, 32) || 'film'}-dialogue.srt`,
  1299→        metaJson: { pipeline: 'short_film', kind: 'dialogue', lines: dialogues.length },
  1300→      });
  1301→    }
  1302→
  1303→    await step.setDetail('চূড়ান্ত MP4 নিবন্ধন');

  1341→        : `কার্যকর, তবে সতর্কতা: ${problems.join(', ')}। ${vInfo.durationSec.toFixed(1)} সেকেন্ড, ${shots.length}টি শট।`;
  1342→    result.quality = verdict;

  1380→/**
  1381→ * S11 — গল্প-টেক্সটকে বাক্য-সীমায় (। . ! ? \n) প্রায়-সমান তিন ভাগে ভাগা।
  1382→ * বাক্য-সীমা না থাকলে অক্ষর-ভিত্তিক সমান তৃতীয়াংশ। কোনো বানানো বিষয়বস্তু নয় —
  1383→ * শুধু প্রদত্ত টেক্সটের নির্ধারক বিভাজন (সৎ-ফলব্যাকের ভিত্তি)।
  1384→ */
  1385→function splitStoryIntoThreeChunks(text: string): [string, string, string] {
  1386→  const clean = (text ?? '').replace(/\s+/g, ' ').trim();
  1387→  if (!clean) return ['সিনেম্যাটিক সূচনা-দৃশ্য', 'গল্পের মাঝামাঝি দৃশ্য', 'সমাপ্তির দৃশ্য'];
  1388→  const sentences = clean
  1389→    .split(/(?<=[।.!?\n])\s+/)
  1390→    .map(s => s.trim())
  1391→    .filter(Boolean);
  1392→  if (sentences.length < 3) {
  1393→    // বাক্য-সীমা নেই — অক্ষর-ভিত্তিক সমান তৃতীয়াংশ
  1394→    const n = Math.ceil(clean.length / 3);
  1395→    return [clean.slice(0, n).trim() || '…', clean.slice(n, 2 * n).trim() || '…', clean.slice(2 * n).trim() || '…'];
  1396→  }
  1397→  const target = clean.length / 3;
  1398→  const buckets = ['', '', ''];
  1399→  let bi = 0;
  1400→  let len = 0;
  1401→  for (const s of sentences) {
  1402→    if (bi < 2 && len >= target) {
  1403→      bi++;
  1404→      len = 0;
  1405→    }
  1406→    buckets[bi] = buckets[bi] ? `${buckets[bi]} ${s}` : s;
  1407→    len += s.length;
  1408→  }
  1409→  return buckets.map((b, idx) => b.trim() || (idx === 0 ? clean : '…')) as [string, string, string];
  1410→}
  1411→
  1412→/**
  1413→ * একাধিক অডিও-খণ্ড নির্দিষ্ট মুহূর্তে বসিয়ে মোট দৈর্ঘ্যের একটি
  1414→ * টাইমলাইন রেন্ডার করে (adelay + amix + apad/atrim)।
  1415→ */
  1416→async function buildPositionedTrack(
  1417→  entries: { file: string; startSec: number; lenSec?: number }[],
  1418→  outputPath: string,
  1419→  totalSec: number
  1420→): Promise<void> {
  1421→  const inputs: string[] = [];
  1422→  const chains: string[] = [];
  1423→  entries.forEach((e, i) => {
  1424→    inputs.push('-i', e.file);
  1425→    const parts = ['aformat=sample_fmts=fltp:sample_rates=44100:channel_layouts=stereo'];
  1426→    if (e.lenSec && e.lenSec > 0) parts.push(`atrim=0:${e.lenSec.toFixed(2)}`, 'asetpts=N/SR/TB');
  1427→    const delay = Math.max(0, Math.round(e.startSec * 1000));
  1428→    if (delay > 0) parts.push(`adelay=${delay}:all=1`);
  1429→    chains.push(`[${i}:a]${parts.join(',')}[t${i}]`);
  1430→  });
  1431→  const labels = entries.map((_, i) => `[t${i}]`).join('');
  1432→  chains.push(
  1433→    `${labels}amix=inputs=${entries.length}:normalize=0,apad=whole_dur=${totalSec.toFixed(2)},atrim=0:${totalSec.toFixed(2)}[out]`
  1434→  );
  1435→  await runFfmpeg(
  1436→    [...inputs, '-filter_complex', chains.join(';'), '-map', '[out]', '-c:a', 'pcm_s16le', outputPath],
  1437→    { timeoutMs: 180_000, label: 'অবস্থানসহ অডিও-টাইমলাইন রেন্ডার' }
  1438→  );
  1439→}
