     3→// song_input → visual_plan → characters → generate_shots →

     5→//
     6→//  - song_input: আপলোড গান → §৬.১ এনার্জি-বিশ্লেষণ (বিট/সেকশন/BPM);
     7→//    গান না থাকলে §৩ প্রসিডিউরাল ইনস্ট্রুমেন্টাল (সৎ লেবেলসহ)।
     8→//  - visual_plan: সেকশন-ভিত্তিক শট-তালিকা — কাট-পয়েন্ট বিট-মার্কার
     9→//    থেকে সমান দূরত্বে বাছাই (প্রতিটি বিট নয়)।
    10→//  - characters: Character-টেবিল থেকে পোর্ট্রেট পুনঃব্যবহার (§৭.২)।
    11→//  - generate_shots: ai_diffusion (শিল্পী-পোর্ট্রেট data-URL দিয়ে i2v)
    12→//    বা cinematic_motion — সৎ-ফলব্যাক ও লেবেলসহ।
    13→//  - timeline: শট-ক্লিপ বিট-কাটে বসানো (ট্রিম+concat; xfade ≤৬),
    14→//    গান মাস্টার-অডিও, ঐচ্ছিক লিরিক-সাবটাইটেল।
    15→//  - final_qc (§৬.৫): দৈর্ঘ্য-মিল / অডিও-গুণমান / A-V সমলয় যাচাই।
    16→//
    17→// সততা: লিপ-সিঙ্ক নেই — "গায়কের মুখ নড়ছে" দাবি করা হয় না;
    18→// প্রসিডিউরাল গান = পরীক্ষামূলক, নিউরাল মিউজিক-মডেল নয়।

    87→export const musicVideoPipeline: PipelineFn = async ctx => {
    88→  const spec = ctx.spec;
    89→  const aspect = spec.aspect ?? '9:16';
    90→  const fps = spec.fps ?? 30;
    91→  const engine = spec.videoEngine === 'cinematic_motion' ? 'cinematic_motion' : 'ai_diffusion';
    92→  const resolution = spec.quality === 'draft' ? '720p' : '1080p';
    93→  const dims = outputDims(aspect, resolution);
    94→  const wd = ctx.workingDir;
    95→  const genSize = ASPECT_TO_GEN_SIZE[aspect];
    96→
    97→  const notes: string[] = [];
    98→
    99→  // ------------------------------------------------------- song_input
   100→  let songPath: string | null = null;
   101→  let songArtifactId: string | null = null;
   102→  let songDuration = 0;
   103→  /** গান থেকে ব্যবহৃত অংশ (টার্গেট-দৈর্ঘ্য) */
   104→  let songUseSec = 0;
   105→  let targetDuration = 0;
   106→  let beatsSec: number[] = [];
   107→  let sections: { name: string; startSec: number; endSec: number; energy: string }[] = [];
   108→  let songBpm = 0;
   109→  let musicSourceLabel = '';
   110→  let analysisLabel = '';
   111→  await ctx.runStep('song_input', 'গান ইনপুট ও বিট-বিশ্লেষণ', async (step: StepHandle) => {
   112→    const desired = spec.durationSec && spec.durationSec > 0 ? spec.durationSec : 60;
   113→    if (spec.songArtifactId) {
   114→      songPath = await ctx.resolveArtifactFile(spec.songArtifactId);
   115→      songArtifactId = spec.songArtifactId;
   116→      const info = await mediaInfo(songPath);
   117→      if (!info.hasAudio) throw new Error('আপলোড করা গানে অডিও স্ট্রিম পাওয়া যায়নি');
   118→      songDuration = info.durationSec;
   119→      // §৬.১ — এনার্জি-বিশ্লেষণ: বিট-মার্কার, সেকশন, আনুমানিক BPM
   120→      const analysis = await analyzeAudioEnergy(songPath);
   121→      beatsSec = analysis.beatsSec;
   122→      sections = analysis.sections.map(s => ({ name: s.energy, startSec: s.startSec, endSec: s.endSec, energy: s.energy }));
   123→      songBpm = analysis.estimatedBpm;
   124→      analysisLabel = analysis.honestLabel;
   125→      targetDuration = Math.max(5, Math.min(desired, songDuration));
   126→      songUseSec = targetDuration;
   127→      musicSourceLabel = 'ব্যবহারকারীর আপলোড করা গান';
   128→      await step.setDetail(
   129→        `আপলোড গান ${songDuration.toFixed(1)} সে. — ≈${songBpm || '?'} BPM, ${beatsSec.length}টি বিট-মার্কার (${analysisLabel})`
   130→      );
   131→    } else if (spec.musicSource === 'none') {
   132→      targetDuration = Math.max(15, Math.min(180, desired));
   133→      musicSourceLabel = 'গান নেই — নীরব টাইমলাইন';
   134→      await step.setDetail('গান ছাড়া নীরব মিউজিক-ভিডিও');
   135→      notes.push('গান নেই — নীরব টাইমলাইন (musicSource: none)।');
   136→    } else {
   137→      // §৩ — প্রসিডিউরাল ইনস্ট্রুমেন্টাল (সৎ লেবেলসহ)
   138→      const moodText = `${spec.mood ?? ''} ${spec.idea ?? ''} ${spec.title ?? ''} ${spec.genre ?? ''}`;
   139→      const mood: MusicMood =
   140→        detectMoodFromText(String(spec.mood ?? '')) ?? detectMoodFromText(moodText) ?? 'romantic';
   141→      const dur = Math.max(8, Math.min(120, desired));
   142→      await step.setDetail(`${moodLabel(mood)} মুডে প্রসিডিউরাল ইনস্ট্রুমেন্টাল (${dur} সে.) সংশ্লেষণ…`);
   143→      const inst = await synthesizeInstrumental({ mood, durationSec: dur, tempoBpm: spec.tempoBpm, genre: spec.genre });
   144→      const wavPath = path.join(wd, 'procedural-song.wav');
   145→      await writeFile(wavPath, inst.buffer);
   146→      const art = await ctx.registerArtifact({
   147→        buffer: inst.buffer,
   148→        kind: 'audio',
   149→        mimeType: 'audio/wav',
   150→        fileName: `${sanitize(spec.title || 'music-video').slice(0, 36)}-instrumental.wav`,
   151→        metaJson: {
   152→          pipeline: 'music_video',
   153→          kind: 'song',
   154→          source: 'procedural',
   155→          mood: inst.meta.mood,
   156→          moodLabel: moodLabel(inst.meta.mood),
   157→          tempoBpm: inst.meta.tempoBpm,
   158→          keyLabel: inst.meta.keyLabel,
   159→          beats: inst.meta.beatsSec.length,
   160→          sections: inst.meta.sections,
   161→          honestLabel: inst.meta.honestLabel,
   162→        },
   163→      });
   164→      songPath = wavPath;
   165→      songArtifactId = art.id;
   166→      songDuration = inst.meta.durationSec;
   167→      beatsSec = inst.meta.beatsSec;
   168→      sections = inst.meta.sections.map(s => ({ name: s.name, startSec: s.startSec, endSec: s.endSec, energy: sectionEnergy(s.name) }));
   169→      songBpm = inst.meta.tempoBpm;
   170→      targetDuration = songDuration;
   171→      songUseSec = songDuration;
   172→      musicSourceLabel = `প্রসিডিউরাল ইনস্ট্রুমেন্টাল (${moodLabel(inst.meta.mood)}, ${inst.meta.tempoBpm} BPM)`;
   173→      notes.push(
   174→        `গান: পরীক্ষামূলক প্রসিডিউরাল ইনস্ট্রুমেন্টাল — নিউরাল মিউজিক-মডেল নয় (${MUSIC_ENGINE_LABEL})।`
   175→      );
   176→      await step.setDetail(
   177→        `প্রসিডিউরাল গান তৈরি — ${inst.meta.durationSec} সে., ${inst.meta.tempoBpm} BPM, ${beatsSec.length}টি বিট-মার্কার`
   178→      );
   179→    }
   180→  });
   181→
   182→  const energyAt = (t: number): string => {
   183→    const s = sections.find(x => t >= x.startSec && t < x.endSec);
   184→    return s?.energy ?? 'medium';
   185→  };
   186→
   187→  // ------------------------------------------------------- visual_plan
   188→  let shots: MvShot[] = [];
   189→  let characters: MvCharacter[] = [];
   190→  let visualStyle = '';
   191→  await ctx.runStep('visual_plan', 'ভিজ্যুয়াল পরিকল্পনা (সেকশন-ভিত্তিক শট-তালিকা)', async (step: StepHandle) => {
   192→    const limit = maxShots('music_video', spec.quality, engine);
   193→    // AI-পথে প্রতি শট ≤৯ সে. (৫/১০-গ্রিড + ট্রানজিশন-প্যাড), সিনেম্যাটিকে ২.৫–৬ সে.
   194→    const segCap = engine === 'ai_diffusion' ? 9 : 6;
   195→    const shotCount = Math.max(engine === 'ai_diffusion' ? 1 : 3, Math.min(limit, Math.ceil(targetDuration / segCap)));
   196→
   197→    // বিট-মার্কার থেকে কাট-পয়েন্ট — সমান দূরত্বে বিট বাছাই (প্রতিটি বিট নয়)
   198→    const usable = beatsSec.filter(b => b > 1 && b < targetDuration - 1);
   199→    const interior: number[] = [];
   200→    if (usable.length >= shotCount - 1) {
   201→      for (let k = 1; k <= shotCount - 1; k++) {
   202→        const idx = Math.min(usable.length - 1, Math.max(0, Math.round((k * usable.length) / shotCount) - 1));
   203→        interior.push(usable[idx]);
   204→      }
   205→    } else {
   206→      for (let k = 1; k <= shotCount - 1; k++) interior.push(round2((k * targetDuration) / shotCount));
   207→    }
   208→    const bounds: number[] = [0];
   209→    for (const b of interior.sort((a, b) => a - b)) {
   210→      if (b - bounds[bounds.length - 1] >= 1.2 && targetDuration - b >= 1.2) bounds.push(round2(b));
   211→    }
   212→    bounds.push(targetDuration);
   213→    const segs = bounds.slice(0, -1).map((b, i) => ({
   214→      startSec: b,
   215→      endSec: bounds[i + 1],
   216→      energy: energyAt((b + bounds[i + 1]) / 2),
   217→    }));
   218→
   219→    const { data } = await chatJson<{
   220→      visualStyle: string;
   221→      characters: { name: string; appearance: string }[];
   222→      shots: { description: string; label: string; shotType: string; cameraMove: string; character?: string | null }[];
   223→    }>([
   224→      {
   225→        role: 'assistant',
   226→        content:
   227→          'তুমি মিউজিক-ভিডিও পরিচালক। গানের সেকশন-শক্তি ও সেগমেন্ট-জানালা অনুযায়ী ঠিক ' +
   228→          `${segs.length}টি শট দাও — প্রতিটি সেগমেন্টের জন্য একটি করে।\n` +
   229→          'শটের shotType: "performance"|"close_up"|"wide"|"establishing" — জোরালো (loud/chorus) অংশে performance/close_up, শান্ত (quiet/intro/outro) অংশে wide/establishing।\n' +
   230→          'description ইংরেজিতে (ভিডিও-জেনারেশনের জন্য দৃশ্যমান বর্ণনা — পরিবেশ, চরিত্র, আলো), label বাংলায়।\n' +
   231→          'cameraMove: "zoom_in"|"zoom_out"|"pan_left"|"pan_right"|"static"।\n' +
   232→          'গায়ক/শিল্পী-চরিত্র দরকার হলে ঠিক ১টি চরিত্র দাও (সংগীতশিল্পী) — appearance ইংরেজিতে বিস্তারিত (বয়স, চেহারা, পোশাক); performance শটগুলোতে ওই চরিত্রের নাম দাও।\n' +
   233→          'শুধু JSON: {"visualStyle":"...","characters":[{"name":"...","appearance":"..."}],"shots":[{"description":"...","label":"...","shotType":"performance","cameraMove":"zoom_in","character":"নাম বা null"}]}',
   234→      },
   235→      {
   236→        role: 'user',
   237→        content:
   238→          `শিরোনাম: ${spec.title}\nআইডিয়া: ${spec.idea ?? '(স্বাধীন)'}\nমোট দৈর্ঘ্য: ${targetDuration.toFixed(1)} সেকেন্ড\n` +
   239→          `গান-উৎস: ${musicSourceLabel}${songBpm ? ` (${songBpm} BPM)` : ''}\n` +
   240→          `সেগমেন্ট-জানালা (শুরু–শেষ, শক্তি):\n${segs.map((s, i) => `  ${i + 1}. ${s.startSec.toFixed(1)}–${s.endSec.toFixed(1)} সে. [${s.energy}]`).join('\n')}\n` +
   241→          (spec.style ? `শৈলী-ইঙ্গিত: ${spec.style}` : ''),
   242→      },
   243→    ]);
   244→
   245→    const rawShots = Array.isArray(data?.shots) ? data.shots : [];
   246→    const typeOk = ['performance', 'close_up', 'wide', 'establishing'];
   247→    shots = segs.map((seg, i) => {
   248→      const s = rawShots[i];
   249→      const description = String(s?.description ?? '').trim();
   250→      return {
   251→        description: (description || fallbackDescription(i, segs.length)).slice(0, 900),
   252→        label: String(s?.label ?? '').slice(0, 120) || `শট ${i + 1}`,
   253→        shotType: (typeOk.includes(String(s?.shotType)) ? s!.shotType : i === 0 ? 'establishing' : 'performance') as MvShotType,
   254→        cameraMove: (MOVES.includes(String(s?.cameraMove) as CameraMove)
   255→          ? (s!.cameraMove as CameraMove)
   256→          : MOVES[i % MOVES.length]) as CameraMove,
   257→        startSec: seg.startSec,
   258→        endSec: seg.endSec,
   259→        durationSec: round2(seg.endSec - seg.startSec),
   260→        character: s?.character ? String(s.character).slice(0, 60) : null,
   261→      };
   262→    });
   263→    if (rawShots.length > segs.length) {
   264→      ctx.partialNotes.push(`LLM ${rawShots.length}টি শট দিয়েছিল — সেগমেন্ট-সীমা অনুযায়ী ${segs.length}টি রাখা হলো।`);
   265→    }
   266→    visualStyle = String(data?.visualStyle ?? spec.style ?? '').slice(0, 200);
   267→
   268→    const rawChars = Array.isArray(data?.characters) ? data.characters : [];
   269→    characters = rawChars
   270→      .filter(c => String(c?.name ?? '').trim() && String(c?.appearance ?? '').trim())
   271→      .slice(0, 1) // সংগীতশিল্পী-চরিত্র ১টি থাকলেই চলে
   272→      .map(c => ({ name: String(c.name).slice(0, 60), appearance: String(c.appearance).slice(0, 600) }));
   273→
   274→    await step.setDetail(
   275→      `${shots.length}টি শট (বিট-কাটে), ${characters.length > 0 ? `চরিত্র: ${characters[0].name}` : 'দৃশ্য-কেন্দ্রিক'}`
   276→    );
   277→  });
   278→
   279→  // ------------------------------------------------------- characters
   280→  await ctx.runStep('characters', 'চরিত্র/পারফরম্যান্স রেফারেন্স', async (step: StepHandle) => {
   281→    if (characters.length === 0) {
   282→      await step.setDetail('দৃশ্য-কেন্দ্রিক ভিজ্যুয়াল — কোনো চরিত্র প্রয়োজন নেই');
   283→      return;
   284→    }
   285→    // §৭.২ — Character-টেবিল থেকে মিল-নাম হলে পোর্ট্রেট পুনঃব্যবহার
   286→    const existing = await db.character.findMany({ where: { conversationId: ctx.conversationId } });
   287→    for (const ch of characters) {
   288→      const match = existing.find(c => c.name.trim() === ch.name || c.name.includes(ch.name) || ch.name.includes(c.name.trim()));
   289→      if (match?.portraitArtifactId) {
   290→        try {
   291→          const p = await ctx.resolveArtifactFile(match.portraitArtifactId);
   292→          ch.portraitDataUrl = await toDataUrl(p);
   293→          ch.portraitPath = p;
   294→          ch.reused = true;
   295→          await step.setDetail(`"${ch.name}" — Character-বাইবেল থেকে পোর্ট্রেট পুনঃব্যবহার`);
   296→          continue;
   297→        } catch {
   298→          // পড়তে ব্যর্থ → নতুন পোর্ট্রেট তৈরি হবে
   299→        }
   300→      }
   301→      await step.setDetail(`"${ch.name}" — পোর্ট্রেট তৈরি`);
   302→      const { buffer } = await generateImageBuffer(portraitPrompt(ch.name, ch.appearance), '864x1152');
   303→      const p = path.join(wd, `character-${sanitize(ch.name) || 'artist'}.png`);
   304→      await writeFile(p, buffer);
   305→      const art = await ctx.registerArtifact({
   306→        buffer,
   307→        kind: 'image',
   308→        mimeType: 'image/png',
   309→        fileName: `character-${sanitize(ch.name) || 'artist'}.png`,
   310→        metaJson: { characterName: ch.name, pipeline: 'music_video', kind: 'artist-reference' },

   663→    result.musicSource = musicSourceLabel;
   664→    result.avSyncOk = avSyncOk;
   665→    result.aspect = aspect;
   666→    result.hasAudio = info.hasAudio;
   667→    result.honestLabel = honestLabel;
   668→    result.notes = notes;
   669→  });
   670→
   671→  return { result };
   672→};

   674→/** সেকশন-নাম → শক্তি-স্তর (প্রসিডিউরাল গানের intro/verse/chorus/outro) */
   675→function sectionEnergy(name: string): string {
   676→  if (name === 'chorus') return 'loud';
   677→  if (name === 'intro' || name === 'outro') return 'quiet';
   678→  return 'medium';
   679→}
